# Notifications: полный MVP-код

## Migration

```sql
CREATE TABLE notification (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    user_id bigint NOT NULL REFERENCES users(id),
    request_id bigint REFERENCES request(id),
    event_id uuid NOT NULL,
    type varchar(50) NOT NULL,
    title varchar(255) NOT NULL,
    body text NOT NULL,
    created_at timestamptz NOT NULL DEFAULT now(),
    read_at timestamptz,
    UNIQUE (event_id, user_id, type)
);
CREATE INDEX idx_notification_user_created ON notification(user_id, created_at DESC);
CREATE INDEX idx_notification_user_unread ON notification(user_id, created_at DESC)
    WHERE read_at IS NULL;
```

## Entity and repository

```java
package ru.miit.servicedesk.notifications;
import java.time.OffsetDateTime;
import java.util.UUID;
import jakarta.persistence.*;
import org.hibernate.annotations.CreationTimestamp;
import ru.miit.servicedesk.requests.entity.RequestEntity;
import ru.miit.servicedesk.users.UserEntity;

@Entity @Table(name = "notification")
public class NotificationEntity {
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY) private Long id;
    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "user_id", nullable = false) private UserEntity user;
    @ManyToOne(fetch = FetchType.LAZY) @JoinColumn(name = "request_id") private RequestEntity request;
    @Column(name = "event_id", nullable = false) private UUID eventId;
    @Column(nullable = false, length = 50) private String type;
    @Column(nullable = false, length = 255) private String title;
    @Column(nullable = false, columnDefinition = "text") private String body;
    @CreationTimestamp @Column(name = "created_at", updatable = false) private OffsetDateTime createdAt;
    @Column(name = "read_at") private OffsetDateTime readAt;
    public Long getId() { return id; } public UserEntity getUser() { return user; }
    public void setUser(UserEntity v) { user = v; } public RequestEntity getRequest() { return request; }
    public void setRequest(RequestEntity v) { request = v; } public UUID getEventId() { return eventId; }
    public void setEventId(UUID v) { eventId = v; } public String getType() { return type; }
    public void setType(String v) { type = v; } public String getTitle() { return title; }
    public void setTitle(String v) { title = v; } public String getBody() { return body; }
    public void setBody(String v) { body = v; } public OffsetDateTime getCreatedAt() { return createdAt; }
    public OffsetDateTime getReadAt() { return readAt; } public void setReadAt(OffsetDateTime v) { readAt = v; }
}
```

```java
package ru.miit.servicedesk.notifications;

import java.time.*;
import org.springframework.data.domain.*;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.*;
import ru.miit.servicedesk.common.NotFoundException;
import ru.miit.servicedesk.security.CurrentUser;

@Service @Transactional(readOnly = true)
public class NotificationQueryService {
    private final NotificationRepository repository; private final CurrentUser currentUser;
    private final Clock clock;
    public NotificationQueryService(NotificationRepository r, CurrentUser c, Clock clock) {
        repository = r; currentUser = c; this.clock = clock;
    }
    public Page<NotificationDto> find(boolean unreadOnly, Pageable pageable) {
        Page<NotificationEntity> page = unreadOnly
                ? repository.findByUser_IdAndReadAtIsNullOrderByCreatedAtDesc(currentUser.id(), pageable)
                : repository.findByUser_IdOrderByCreatedAtDesc(currentUser.id(), pageable);
        return page.map(this::dto);
    }
    public long unreadCount() { return repository.countByUser_IdAndReadAtIsNull(currentUser.id()); }
    @Transactional public NotificationDto markRead(Long id) {
        NotificationEntity n = repository.findByIdAndUser_Id(id, currentUser.id())
                .orElseThrow(() -> new NotFoundException("Notification not found"));
        if (n.getReadAt() == null) n.setReadAt(now());
        return dto(n);
    }
    @Transactional public void markAllRead() { repository.markAllRead(currentUser.id(), now()); }
    private OffsetDateTime now() { return OffsetDateTime.ofInstant(clock.instant(), ZoneOffset.UTC); }
    private NotificationDto dto(NotificationEntity n) {
        return new NotificationDto(n.getId(), n.getType(), n.getTitle(), n.getBody(),
                n.getRequest() == null ? null : n.getRequest().getId(), n.getCreatedAt(), n.getReadAt());
    }
}
```

```java
package ru.miit.servicedesk.notifications;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.*;
import org.springframework.transaction.event.TransactionPhase;
import org.springframework.transaction.event.TransactionalEventListener;
import ru.miit.servicedesk.users.UserRepository;
import ru.miit.servicedesk.workflow.RequestStatusChangedEvent;

@Component
public class RequestNotificationListener {
    private final NotificationRepository notifications; private final UserRepository users;
    private final ru.miit.servicedesk.requests.RequestRepository requests;
    public RequestNotificationListener(NotificationRepository n, UserRepository u,
            ru.miit.servicedesk.requests.RequestRepository r) { notifications = n; users = u; requests = r; }
    @Async
    @Transactional(propagation = Propagation.REQUIRES_NEW)
    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    public void onStatusChanged(RequestStatusChangedEvent event) {
        var request = requests.findDetailedById(event.requestId()).orElseThrow();
        var recipient = request.getAuthor();
        NotificationEntity n = new NotificationEntity();
        n.setEventId(event.eventId()); n.setUser(users.getReferenceById(recipient.getId()));
        n.setRequest(request); n.setType("REQUEST_STATUS_CHANGED");
        n.setTitle("Статус заявки изменён");
        n.setBody(event.previousStatus() + " → " + event.newStatus());
        notifications.save(n);
    }
}
```

```java
package ru.miit.servicedesk.notifications;
import org.springframework.data.domain.Pageable;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import ru.miit.servicedesk.common.PageResponse;
@RestController @RequestMapping("/api/v1/notifications")
public class NotificationController {
    private final NotificationQueryService service;
    public NotificationController(NotificationQueryService s) { service = s; }
    @GetMapping PageResponse<NotificationDto> find(@RequestParam(defaultValue = "false") boolean unreadOnly,
            Pageable p) { return PageResponse.from(service.find(unreadOnly, p), v -> v); }
    @GetMapping("/unread-count") long count() { return service.unreadCount(); }
    @PutMapping("/{id}/read") NotificationDto read(@PathVariable Long id) { return service.markRead(id); }
    @PutMapping("/read-all") ResponseEntity<Void> all() { service.markAllRead(); return ResponseEntity.noContent().build(); }
}
```
