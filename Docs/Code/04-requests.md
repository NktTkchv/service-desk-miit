## `requests/RequestMapper.java`

```java
package ru.miit.servicedesk.requests;

import ru.miit.servicedesk.requests.dto.RequestDto;
import ru.miit.servicedesk.users.UserMapper;

public final class RequestMapper {
    private RequestMapper() {
    }

    public static RequestDto toDto(RequestEntity entity) {
        return new RequestDto(
                entity.getId(),
                UserMapper.toSummary(entity.getAuthor()),
                entity.getAudience().getId(),
                entity.getAudience().getNumber(),
                entity.getService().getId(),
                entity.getService().getNameService(),
                entity.getStatus().getName(),
                entity.getDescription(),
                entity.getApprover() == null ? null : UserMapper.toSummary(entity.getApprover()),
                entity.getExecutor() == null ? null : UserMapper.toSummary(entity.getExecutor()),
                entity.getCreatedAt(),
                entity.getDueAt(),
                entity.getVersion()
        );
    }
}
```

## `requests/RequestServiceImpl.java`

```java
package ru.miit.servicedesk.requests;

import java.time.Clock;
import java.time.OffsetDateTime;
import java.time.ZoneOffset;

import org.springframework.context.ApplicationEventPublisher;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import ru.miit.servicedesk.catalog.entity.ServiceCatalogEntity;
import ru.miit.servicedesk.catalog.ServiceCatalogRepository;
import ru.miit.servicedesk.common.NotFoundException;
import ru.miit.servicedesk.requests.dto.CreateRequestRequest;
import ru.miit.servicedesk.requests.dto.RequestDto;
import ru.miit.servicedesk.security.CurrentUser;
import ru.miit.servicedesk.users.UserEntity;
import ru.miit.servicedesk.users.UserRepository;

@Service
@Transactional(readOnly = true)
public class RequestServiceImpl implements RequestService {
    private final RequestRepository requestRepository;
    private final AudienceRepository audienceRepository;
    private final ServiceCatalogRepository catalogRepository;
    private final RequestStatusRepository statusRepository;
    private final UserRepository userRepository;
    private final CurrentUser currentUser;
    private final Clock clock;
    private final ApplicationEventPublisher events;

    public RequestServiceImpl(
            RequestRepository requestRepository, AudienceRepository audienceRepository,
            ServiceCatalogRepository catalogRepository, RequestStatusRepository statusRepository,
            UserRepository userRepository, CurrentUser currentUser, Clock clock,
            ApplicationEventPublisher events
    ) {
        this.requestRepository = requestRepository;
        this.audienceRepository = audienceRepository;
        this.catalogRepository = catalogRepository;
        this.statusRepository = statusRepository;
        this.userRepository = userRepository;
        this.currentUser = currentUser;
        this.clock = clock;
        this.events = events;
    }

    @Override
    @Transactional
    public RequestDto create(CreateRequestRequest command) {
        UserEntity author = userRepository.findById(currentUser.id())
                .orElseThrow(() -> new NotFoundException("Current user not found"));
        AudienceEntity audience = audienceRepository.findById(command.audienceId())
                .orElseThrow(() -> new NotFoundException("Audience not found"));
        ServiceCatalogEntity service = catalogRepository.findByIdAndActiveTrue(command.serviceId())
                .orElseThrow(() -> new NotFoundException("Active service not found"));
        RequestStatusEntity status = statusRepository.findByName(RequestStatus.PENDING_HEAD.name())
                .orElseThrow(() -> new IllegalStateException("PENDING_HEAD status is not configured"));
        OffsetDateTime now = OffsetDateTime.ofInstant(clock.instant(), ZoneOffset.UTC);

        RequestEntity entity = new RequestEntity();
        entity.setAuthor(author);
        entity.setAudience(audience);
        entity.setService(service);
        entity.setStatus(status);
        entity.setDescription(command.description());
        entity.setDeadlineHoursSnapshot(service.getDeadline());
        entity.setDueAt(now.plusHours(service.getDeadline()));
        entity = requestRepository.save(entity);
        events.publishEvent(new RequestCreatedEvent(entity.getId(), author.getId(), clock.instant()));
        return RequestMapper.toDto(entity);
    }

    @Override
    public RequestDto getById(Long id) {
        RequestEntity entity = requestRepository.findDetailedById(id)
                .orElseThrow(() -> new NotFoundException("Request not found: " + id));
        requireVisible(entity);
        return RequestMapper.toDto(entity);
    }

    @Override
    public Page<RequestDto> findVisible(Pageable pageable) {
        return switch (currentUser.role()) {
            case "ADMIN" -> requestRepository.findAll(pageable).map(RequestMapper::toDto);
            case "TEACHER" -> requestRepository.findByAuthor_Id(currentUser.id(), pageable)
                    .map(RequestMapper::toDto);
            case "DEPARTMENT_HEAD", "DEPARTMENT_STAFF" -> requestRepository
                    .findByAuthor_Department_Id(currentUser.departmentId().orElseThrow(), pageable)
                    .map(RequestMapper::toDto);
            case "DUTY_OFFICER" -> requestRepository.findByExecutor_Id(currentUser.id(), pageable)
                    .map(RequestMapper::toDto);
            default -> Page.empty(pageable);
        };
    }

    private void requireVisible(RequestEntity entity) {
        boolean visible = switch (currentUser.role()) {
            case "ADMIN" -> true;
            case "TEACHER" -> entity.getAuthor().getId().equals(currentUser.id());
            case "DEPARTMENT_HEAD", "DEPARTMENT_STAFF" -> entity.getAuthor().getDepartment() != null
                    && entity.getAuthor().getDepartment().getId()
                            .equals(currentUser.departmentId().orElse(null));
            case "DUTY_OFFICER" -> entity.getExecutor() != null
                    && entity.getExecutor().getId().equals(currentUser.id());
            default -> false;
        };
        if (!visible) throw new NotFoundException("Request not found: " + entity.getId());
    }
}
```

## `requests/RequestController.java`

```java
package ru.miit.servicedesk.requests;

import java.net.URI;
import jakarta.validation.Valid;
import org.springframework.data.domain.Pageable;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import ru.miit.servicedesk.common.PageResponse;
import ru.miit.servicedesk.requests.dto.CreateRequestRequest;
import ru.miit.servicedesk.requests.dto.RequestDto;

@RestController
@RequestMapping("/api/v1/requests")
public class RequestController {
    private final RequestService service;
    public RequestController(RequestService service) { this.service = service; }

    @PostMapping
    ResponseEntity<RequestDto> create(@Valid @RequestBody CreateRequestRequest request) {
        RequestDto dto = service.create(request);
        return ResponseEntity.created(URI.create("/api/v1/requests/" + dto.id())).body(dto);
    }

    @GetMapping("/{id}")
    RequestDto get(@PathVariable Long id) { return service.getById(id); }

    @GetMapping
    PageResponse<RequestDto> find(Pageable pageable) {
        return PageResponse.from(service.findVisible(pageable), value -> value);
    }
}
```

Для фильтров из REST-спецификации этот MVP расширяется `JpaSpecificationExecutor`: role scope обязательно добавляется к specification через `and`, а не заменяется пользовательским фильтром.
