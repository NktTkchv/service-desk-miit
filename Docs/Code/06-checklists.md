```sql
CREATE TABLE checklist_template_item (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    service_id bigint NOT NULL REFERENCES service_catalog(id),
    title varchar(255) NOT NULL,
    position integer NOT NULL CHECK (position >= 0),
    required boolean NOT NULL DEFAULT true,
    active boolean NOT NULL DEFAULT true,
    UNIQUE (service_id, position)
);
CREATE TABLE request_checklist_item (
    id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    request_id bigint NOT NULL REFERENCES request(id),
    template_item_id bigint REFERENCES checklist_template_item(id),
    title varchar(255) NOT NULL,
    position integer NOT NULL,
    required boolean NOT NULL,
    completed boolean NOT NULL DEFAULT false,
    completed_by bigint REFERENCES users(id),
    completed_at timestamptz,
    UNIQUE (request_id, position),
    CHECK ((completed = false AND completed_at IS NULL AND completed_by IS NULL)
        OR (completed = true AND completed_at IS NOT NULL AND completed_by IS NOT NULL))
);
CREATE INDEX idx_checklist_template_service ON checklist_template_item(service_id, active, position);
CREATE INDEX idx_checklist_request ON request_checklist_item(request_id, position);
```

```java
package ru.miit.servicedesk.checklists;

import java.time.*;
import java.util.List;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import ru.miit.servicedesk.common.NotFoundException;
import ru.miit.servicedesk.requests.entity.RequestEntity;
import ru.miit.servicedesk.security.CurrentUser;
import ru.miit.servicedesk.users.UserRepository;

@Service @Transactional(readOnly = true)
public class ChecklistServiceImpl implements ChecklistService {
    private final ChecklistTemplateItemRepository templates;
    private final RequestChecklistItemRepository items;
    private final UserRepository users;
    private final CurrentUser currentUser;
    private final Clock clock;
    public ChecklistServiceImpl(ChecklistTemplateItemRepository templates,
            RequestChecklistItemRepository items, UserRepository users,
            CurrentUser currentUser, Clock clock) {
        this.templates = templates; this.items = items; this.users = users;
        this.currentUser = currentUser; this.clock = clock;
    }
    @Transactional
    public void createSnapshotForRequest(RequestEntity request) {
        List<RequestChecklistItemEntity> snapshot = templates
                .findByService_IdAndActiveTrueOrderByPosition(request.getService().getId()).stream()
                .map(template -> copy(request, template)).toList();
        items.saveAll(snapshot);
    }
    public List<ChecklistItemDto> getForRequest(Long id) {
        return items.findByRequest_IdOrderByPosition(id).stream().map(this::dto).toList();
    }
    @Transactional
    public ChecklistItemDto setCompleted(Long id, boolean completed) {
        RequestChecklistItemEntity item = items.findDetailedById(id)
                .orElseThrow(() -> new NotFoundException("Checklist item not found"));
        if (item.getRequest().getExecutor() == null
                || !item.getRequest().getExecutor().getId().equals(currentUser.id()))
            throw new AccessDeniedException("Only assigned executor can update checklist");
        if (item.isCompleted() != completed) {
            item.setCompleted(completed);
            item.setCompletedBy(completed ? users.getReferenceById(currentUser.id()) : null);
            item.setCompletedAt(completed
                    ? OffsetDateTime.ofInstant(clock.instant(), ZoneOffset.UTC) : null);
        }
        return dto(item);
    }
    public boolean areRequiredItemsComplete(Long id) { return items.areRequiredComplete(id); }
    private RequestChecklistItemEntity copy(RequestEntity r, ChecklistTemplateItemEntity t) {
        RequestChecklistItemEntity i = new RequestChecklistItemEntity();
        i.setRequest(r); i.setTemplateItem(t); i.setTitle(t.getTitle());
        i.setPosition(t.getPosition()); i.setRequired(t.isRequired()); return i;
    }
    private ChecklistItemDto dto(RequestChecklistItemEntity i) {
        return new ChecklistItemDto(i.getId(), i.getTitle(), i.getPosition(), i.isRequired(),
                i.isCompleted(), i.getCompletedBy() == null ? null : i.getCompletedBy().getId(),
                i.getCompletedAt());
    }
}
```
